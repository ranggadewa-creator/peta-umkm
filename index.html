html,body{ margin:0; padding:0; height:100%; font-family:Arial; } #header{ background:#1565C0; color:white; padding:15px; text-align:center; } #map{ height:92vh; } .popup img{ width:250px; border-radius:10px; } .legend{ background:white; padding:10px; border-radius:8px; line-height:28px; box-shadow:0 0 8px rgba(0,0,0,.3); } //------------------------------------ // BASEMAP //------------------------------------ var osm = L.tileLayer( 'https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', { maxZoom:19, attribution:'© OpenStreetMap' } ); var satelit = L.tileLayer( 'https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/{z}/{y}/{x}', { attribution:'Esri' } ); //------------------------------------ // MAP //------------------------------------ var map = L.map('map',{ center:[1.70,128.02], zoom:15, layers:[osm] }); //------------------------------------ // BACA BATAS DESA //------------------------------------ var batasDesa; fetch("data/batas_desa.geojson") .then(res=>res.json()) .then(data=>{ batasDesa=L.geoJSON(data,{ style:{ color:"green", weight:2, fillOpacity:0.05 } }).addTo(map); }); //------------------------------------ // BACA UMKM //------------------------------------ var umkmLayer; fetch("data/UMKM_Geojson.geojson") .then(res=>res.json()) .then(data=>{ umkmLayer=L.geoJSON(data,{ pointToLayer:function(feature,latlng){ return L.marker(latlng); }, onEachFeature:function(feature,layer){ let p=feature.properties; layer.bindPopup(`

${p["Nama Usaha"]}

Pemilik
${p.Pemilik}

Deskripsi
${p.Deskripsi}

Lihat di Google Maps
`); } }).addTo(map); // SEARCH var search = new L.Control.Search({ layer:umkmLayer, propertyName:'Nama Usaha', zoom:18 }); map.addControl(search); // LAYER CONTROL L.control.layers( { "OpenStreetMap":osm, "Satelit":satelit }, { "UMKM":umkmLayer, "Batas Desa":batasDesa } ).addTo(map); }); //------------------------------------ // LEGENDA //------------------------------------ var legend=L.control({ position:'bottomright' }); legend.onAdd=function(){ var div=L.DomUtil.create('div','legend'); div.innerHTML=` Legenda

🔴 UMKM
🟢 Batas Desa `; return div; }; legend.addTo(map);
